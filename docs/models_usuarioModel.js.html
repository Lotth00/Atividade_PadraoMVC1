/**
 * Modelo de dados para usuários (armazenamento em memória).
 * @module models/usuarioModel
 */

/**
 * @typedef {Object} Usuario
 * @property {number} id - ID do usuário.
 * @property {string} nome - Nome completo.
 * @property {string} email - Email único.
 * @property {string} senha - Senha (armazenada em texto puro neste exemplo).
 * @property {string} tipo - Tipo de usuário: 'admin' ou 'usuario'.
 */

/** @type {Usuario[]} */
const usuarios = [
    {
        id: 1,
        nome: 'Administrador',
        email: 'admin@teste.com',
        senha: '123456',
        tipo: 'admin'
    },
    // outros usuários podem ser adicionados dinamicamente
];

const usuarioModel = {
    /**
     * Retorna todos os usuários, removendo o campo `senha`.
     * @returns {Array<Omit<Usuario, 'senha'>>} Lista de usuários sem a senha.
     */
    findAll: () => {
        return usuarios.map(({ senha, ...usuario }) => usuario);
    },
    
    /**
     * Busca um usuário pelo email.
     * @param {string} email - Email do usuário.
     * @returns {Usuario | undefined} Usuário completo (com senha) ou undefined.
     */
    findByEmail: (email) => {
        return usuarios.find(usuario => usuario.email === email);
    },
    
    /**
     * Busca um usuário pelo ID, retornando seus dados sem a senha.
     * @param {number | string} id - ID do usuário.
     * @returns {Omit<Usuario, 'senha'> | null} Usuário sem senha ou null.
     */
    findById: (id) => {
        const usuario = usuarios.find(u => u.id === parseInt(id));
        if (usuario) {
            const { senha, ...usuarioSemSenha } = usuario;
            return usuarioSemSenha;
        }
        return null;
    },
    
    /**
     * Verifica se o email e senha correspondem a um usuário.
     * @param {string} email - Email fornecido.
     * @param {string} senha - Senha fornecida.
     * @returns {Omit<Usuario, 'senha'> | null} Usuário sem senha se autenticado, ou null.
     */
    verificarLogin: (email, senha) => {
        const usuario = usuarios.find(u => u.email === email && u.senha === senha);
        if (usuario) {
            const { senha, ...usuarioSemSenha } = usuario;
            return usuarioSemSenha;
        }
        return null;
    },
    
    /**
     * Cria um novo usuário.
     * @param {Omit<Usuario, 'id'>} usuario - Dados do novo usuário (sem id).
     * @returns {Omit<Usuario, 'senha'>} Usuário criado, sem a senha.
     */
    create: (usuario) => {
        const novoUsuario = {
            id: usuarios.length + 1,
            ...usuario
        };
        usuarios.push(novoUsuario);
        const { senha, ...usuarioSemSenha } = novoUsuario;
        return usuarioSemSenha;
    },
    
    /**
     * Atualiza os dados de um usuário existente.
     * @param {number | string} id - ID do usuário a ser atualizado.
     * @param {Partial<Usuario>} dadosAtualizados - Dados parciais para atualizar.
     * @returns {Omit<Usuario, 'senha'> | null} Usuário atualizado sem senha, ou null se não encontrado.
     */
    update: (id, dadosAtualizados) => {
        const index = usuarios.findIndex(u => u.id === parseInt(id));
        if (index !== -1) {
            usuarios[index] = { ...usuarios[index], ...dadosAtualizados };
            const { senha, ...usuarioSemSenha } = usuarios[index];
            return usuarioSemSenha;
        }
        return null;
    },
    
    /**
     * Remove um usuário pelo ID.
     * @param {number | string} id - ID do usuário.
     * @returns {boolean} True se removido, false se não encontrado.
     */
    delete: (id) => {
        const index = usuarios.findIndex(u => u.id === parseInt(id));
        if (index !== -1) {
            usuarios.splice(index, 1);
            return true;
        }
        return false;
    }
};

module.exports = usuarioModel;
