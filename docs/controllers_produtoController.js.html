};

/**
 * Exibe a página inicial (home) com a lista de produtos.
 * ⚠️ Atenção: esta função sobrescreve a anterior `getHome`.
 * @param {Request} req - Objeto da requisição.
 * @param {Response} res - Renderiza home com produtos.
 * @returns {void}
 * @route GET /
 * @access Public
 */
exports.getHome = (req, res) => {
    res.render('home', { produtos }); 
};

/**
 * Cria um novo produto a partir dos dados do formulário.
 * @param {Request} req - Corpo da requisição com { nome, preco }.
 * @param {Response} res - Redireciona para a lista de produtos.
 * @returns {void}
 * @route POST /criar-produto
 * @access Private (usuário autenticado)
 */
exports.criarProduto = (req, res) => {
    let ultimoId = 0;
    
    if (produtos.length > 0) {
        ultimoId = produtos[produtos.length - 1].id;
    }

    const novoId = ultimoId + 1;

    const novo = { 
        id: novoId, 
        nome: req.body.nome, 
        preco: req.body.preco 
    };

    produtos.push(novo);
    res.redirect('/produtos');
};

/**
 * Deleta um produto pelo ID.
 * @param {Request} req - Parâmetro da URL com { id }.
 * @param {Response} res - Redireciona para a lista de produtos.
 * @returns {void}
 * @route POST /deletar-produto/:id
 * @access Private (usuário autenticado)
 */
exports.deletarProduto = (req, res) => {
    const id = req.params.id;
    produtos = produtos.filter(p => p.id != id);
    res.redirect('/produtos');
};

/**
 * Exibe o formulário de edição de um produto.
 * @param {Request} req - Parâmetro da URL com { id }.
 * @param {Response} res - Renderiza a página editarProdutos com os dados do produto.
 * @returns {void}
 * @route GET /editar-produto/:id
 * @access Private (usuário autenticado)
 */
exports.getEditarProduto = (req, res) => {
    const id = req.params.id;
    const produto = produtos.find(p => p.id == id);
    res.render('editarProdutos', { produto });
};

/**
 * Salva as alterações de um produto após edição.
 * @param {Request} req - Parâmetro { id } e corpo { nome, preco }.
 * @param {Response} res - Redireciona para a lista de produtos.
 * @returns {void}
 * @route POST /editar-produto/:id
 * @access Private (usuário autenticado)
 */
exports.postEditarProduto = (req, res) => {
    const id = req.params.id;
    const indice = produtos.findIndex(p => p.id == id);

    if (indice !== -1) {
        produtos[indice].nome = req.body.nome;
        produtos[indice].preco = req.body.preco;
    }
    res.redirect('/produtos');
};
