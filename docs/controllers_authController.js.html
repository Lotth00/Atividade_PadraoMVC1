/**
 * Controlador de autenticação e gestão de usuários.
 * @module controllers/authController
 */

const usuarioModel = require('../models/usuarioModel');

/**
 * @typedef {import('express').Request} Request
 * @typedef {import('express').Response} Response
 */

const authController = {
    /**
     * Exibe a tela de login.
     * Se o usuário já estiver logado, redireciona para a home.
     * @param {Request} req - Objeto da requisição Express.
     * @param {Response} res - Objeto de resposta Express.
     * @returns {void}
     * @route GET /login
     * @access Public
     */
    telaLogin: (req, res) => {
        if (req.session && req.session.user) {
            return res.redirect('/');
        }
        res.render('login');
    },

    /**
     * Processa o formulário de login.
     * Verifica credenciais e cria sessão se bem-sucedido.
     * @param {Request} req - Corpo da requisição com { email, senha }.
     * @param {Response} res - Resposta com redirecionamento ou mensagem de erro.
     * @returns {void}
     * @route POST /login
     * @access Public
     */
    processarLogin: (req, res) => {
        const { email, senha } = req.body;
        
        const usuario = usuarioModel.verificarLogin(email, senha);
        
        if (usuario) {
            req.session.user = usuario;
            console.log(`Login realizado: ${usuario.nome} (${usuario.email})`);
            res.redirect('/');
        } else {
            res.send(`
                <script>
                    alert('Email ou senha incorretos!');
                    window.location.href = '/login';
                </script>
            `);
        }
    },

    /**
     * Realiza logout do usuário.
     * Destrói a sessão e redireciona para o login.
     * @param {Request} req - Objeto da requisição Express.
     * @param {Response} res - Resposta com redirecionamento.
     * @returns {void}
     * @route GET /logout
     * @access Private (usuário autenticado)
     */
    logout: (req, res) => {
        req.session.destroy((erro) => {
            if (erro) {
                console.error('Erro no logout:', erro);
                res.send('Erro ao fazer logout');
            } else {
                res.redirect('/login');
            }
        });
    },
    
    /**
     * Lista todos os usuários (apenas para admin).
     * @param {Request} req - Objeto da requisição.
     * @param {Response} res - Renderiza a página de usuários ou mensagem de acesso negado.
     * @returns {void}
     * @route GET /usuarios
     * @access Private (apenas admin)
     */
    listarUsuarios: (req, res) => {
        if (req.session.user.tipo !== 'admin') {
            return res.send('Acesso negado! Apenas administradores.');
        }
        
        const usuarios = usuarioModel.findAll();
        res.render('usuarios', { 
            usuarios: usuarios,
            usuario: req.session.user 
        });
    },

    /**
     * Exibe o formulário de criação de novo usuário (apenas admin).
     * @param {Request} req - Objeto da requisição.
     * @param {Response} res - Renderiza o formulário ou mensagem de acesso negado.
     * @returns {void}
     * @route GET /usuarios/novo
     * @access Private (apenas admin)
     */
    formCriarUsuario: (req, res) => {
        if (req.session.user.tipo !== 'admin') {
            return res.send('Acesso negado! Apenas administradores.');
        }
        res.render('criarUsuario', { usuario: req.session.user });
    },
    
    /**
     * Cria um novo usuário (apenas admin).
     * @param {Request} req - Corpo da requisição com { nome, email, senha, tipo }.
     * @param {Response} res - Redireciona após criação ou exibe erro.
     * @returns {void}
     * @route POST /usuarios/novo
     * @access Private (apenas admin)
     */
    criarUsuario: (req, res) => {
        if (req.session.user.tipo !== 'admin') {
            return res.send('Acesso negado! Apenas administradores.');
        }
        
        const { nome, email, senha, tipo } = req.body;
    
        const usuarioExistente = usuarioModel.findByEmail(email);
        if (usuarioExistente) {
            return res.send(`
                <script>
                    alert('Email já cadastrado!');
                    window.location.href = '/usuarios/novo';
                </script>
            `);
        }
        
        const novoUsuario = usuarioModel.create({
            nome,
            email,
            senha,
            tipo: tipo || 'usuario'
        });
        
        res.redirect('/usuarios');
    },
    
    /**
     * Deleta um usuário por ID (apenas admin).
     * @param {Request} req - Parâmetro da URL com { id }.
     * @param {Response} res - Redireciona após deletar.
     * @returns {void}
     * @route POST /deletar-usuario/:id
     * @access Private (apenas admin)
     */
    deletarUsuario: (req, res) => {
        if (req.session.user.tipo !== 'admin') {
            return res.send('Acesso negado! Apenas administradores.');
        }
        
        const { id } = req.params;
        usuarioModel.delete(id);
        res.redirect('/usuarios');
    }
};

module.exports = authController;
